    int len;

        perror("Failed to open");
        return 1;
    }

    printf("Enter message:\n");

    nr_read = read(0, input, sizeof(input));
    if (nr_read < 0) {
        perror("Failed to read");

        if (input[nr_read - 1] == '\n') {

        perror("Failed to write");
